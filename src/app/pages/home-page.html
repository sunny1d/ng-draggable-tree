快速上手

// 1. 构建库并发布 / 安装（见根目录 package.json scripts.publish）
npm run build:lib   # ng build ng-draggable-tree

// 2. 在组件中引入（本示例仓库通过 tsconfig paths 指向库源码 public-api）
import { NgDraggableTreeComponent, type TreeOptions } from 'ng-draggable-tree';
import { Component, signal } from '@angular/core';

interface Node { id: string; name: string; children?: Node[]; }

@Component({
  selector: 'app-demo',
  imports: [NgDraggableTreeComponent],
  template: '<ng-draggable-tree [(nodes)]="nodes" [options]="options" />',
})
export class Demo {
  readonly nodes = signal<Node[]>([
    { id: '1', name: '项目', children: [{ id: '1-1', name: 'src' }] },
    { id: '2', name: '文档' },
  ]);
  readonly options: TreeOptions<Node> = {
    idField: 'id',
    displayField: 'name',
    childrenField: 'children',
    hasChildrenField: 'hasChildren',
    showLine: true,
  };
}

更多 API（懒加载 / 复选框 / 拖拽 / 事件回调等）请查阅 projects/ng-draggable-tree/README.md 与各示例页面侧栏说明。
